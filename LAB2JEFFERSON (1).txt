public class Mavenproject1 {

    public static void main(String[] args) {
        Scanner ler=new Scanner(System.in);
        int[][] m=new int[4][4];
        double somadp=0, somads=0;
        
        for(int i=0;i<4;i++){
            for(int j=0;j<4;j++){
                System.out.println("DIGITE O VALOR DA POSICAO["+i+j+"]:");
                m[i][j]=ler.nextInt();
            }
        }
        for(int i=0;i<4;i++){
            for(int j=0;j<4;j++){
                if(i==j){
                    System.out.println("DP: "+m[i][j]); 
                    somadp+=m[i][j];
                }
            }
        }
        for(int i=0;i<4;i++){
            for(int j=0;j<4;j++){
                if(i+j==3){
                    System.out.println("DS: "+m[i][j]); 
                    somads+=m[i][j];
                }
            }
        }
        System.out.println("SOMA DP:"+somadp);
        System.out.println("SOMA DS:"+somads);
        
        double maior;
        if(somadp<somads){
            maior=somadp;
        } else if(somads<somadp){
            maior=somads;
        }else if(somadp==somads){
            System.out.println("AS DUAS SAO IGUAIS");
        }
    }
}

public static void main(String[] args) {
    Scanner ler=new Scanner(System.in);
    int[][] m=new int[3][3];
    int[] cont=new int[10];
    int erros=0;

    for(int i=0;i<3;i++){
        for(int j=0;j<3;j++){
            System.out.println("digite o valor:");
            m[i][j]=ler.nextInt();
            if(m[i][j]<1 || m[i][j]>9){
                erros++;
            }else{
                cont[m[i][j]]++;
            }
        }
    }

    for(int i=1;i<=9;i++){
        if(cont[i]!=1){
            System.out.println("numero repetido ou ausente:"+i);
            erros++;
        }
    }

    int somaref=0;
    for(int j=0;j<3;j++){
        somaref+=m[0][j];
    }

    for(int i=1;i<3;i++){
        int soma=0;
        for(int j=0;j<3;j++){
            soma+=m[i][j];
        }
        if(soma!=somaref){
            System.out.println("linha diferente:"+i);
            erros++;
        }
    }

    for(int j=0;j<3;j++){
        int soma=0;
        for(int i=0;i<3;i++){
            soma+=m[i][j];
        }
        if(soma!=somaref){
            System.out.println("coluna diferente:"+j);
            erros++;
        }
    }

    int d1=0;
    int d2=0;

    for(int i=0;i<3;i++){
        d1+=m[i][i];
        d2+=m[i][2-i];
    }

    if(d1!=somaref || d2!=somaref){
        System.out.println("diagonais diferentes");
        erros++;
    }

    if(erros==0){
        System.out.println("a matriz e magica!");
    }else{
        System.out.println("a matriz nao e magica!");
    }
}

public static void main(String[] args) {
    Scanner ler=new Scanner(System.in);
        char[][] tab=new char[8][8];
        char[][] vis=new char[8][8];
        int linha,coluna;
        
        for(int i=0;i<8;i++){
            for(int j=0;j<8;j++){
                tab[i][j]='A';
                vis[i][j]='A';
            }
        }
        int totalNav=5;
        int NavColoc=0;
        while(NavColoc<totalNav){
            System.out.println("navio "+(NavColoc+1)+" de "+totalNav);
            System.out.println("linha (0 a 7):");
            linha=ler.nextInt();
            System.out.println("coluna (0 a 7):");
            coluna=ler.nextInt();
            
            if(linha<0||linha>7||coluna<0||coluna>7){
                System.out.println("posiçao fora do tabuleiro! tente novamente!");
                continue;
            }
            if(tab[linha][coluna]=='N'){
                System.out.println("posicao ja ocupada! tente novamente!");
                continue;
            }
            
            tab[linha][coluna]='N';
            NavColoc++;
            System.out.println("navio posicionado!"); 
        }
        int Maxdisp=15;
        int acerto=0;
        int erro=0;
        int dispRepetido=0;
        int NavAfund=0;
      
        
        System.out.println("voce tem "+Maxdisp+" disparos!");
        for(int disp=0;disp<15;disp++){
            System.out.println(" 0 1 2 3 4 5 6 7 ");
            for(int i=0;i<8;i++){
                System.out.print(i+" ");
                for(int j=0;j<8;j++){
                    System.out.print(vis[i][j]+" ");
                }
                System.out.println();
            }
             System.out.println("\ndisparo "+disp+" de "+Maxdisp);
             System.out.println("linha (0 a 7):");
             linha=ler.nextInt();
             System.out.println("coluna (0 a 7):");
             coluna=ler.nextInt();
             
              if(linha<0||linha>7||coluna<0||coluna>7){
                System.out.println("posiçao fora do tabuleiro! disparo perdido!");
                erro++;
                continue;
              }
              if(vis[linha][coluna]=='X'||vis[linha][coluna]=='E'){
                  System.out.println("disparo repetido!");
                  dispRepetido++;
                  continue;
              }
              
              if(tab[linha][coluna]=='N'){
                  System.out.println("acertou um navio!");
                  vis[linha][coluna]='X';
                  acerto++;
                  NavAfund++;
              }else{
                  System.out.println("errou!");
                  vis[linha][coluna]='E';
                  erro++;
              }
              
              if(NavAfund==totalNav){
                  System.out.println("voce afundou todos!parabens!");
                  break;
              }
        }
        System.out.println("fim");
        System.out.println("Acertos: " + acerto);
        System.out.println("Erros:  " + erro);
        System.out.println("Disparos repetidos: " + dispRepetido);
        
        if (NavAfund<totalNav) {
            System.out.println("Navios restantes: " + (totalNav - NavAfund));
            System.out.println("Você perdeu!");
        }

    }
    
}
